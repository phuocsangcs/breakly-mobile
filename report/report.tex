% 2x2 SWOT grid: \swotbox{S items}{W items}{O items}{T items}
% (each argument is a sequence of \item ... lines)
\newcommand{\swotbox}[4]{%
  \begin{longtable}{|p{0.47\textwidth}|p{0.47\textwidth}|}
    \hline
    \textbf{Strengths} & \textbf{Weaknesses} \\ \hline
    \endfirsthead
    \multicolumn{2}{r}{\textit{Continued from previous page}}\\[-2pt]\hline
    \textbf{Strengths} & \textbf{Weaknesses} \\ \hline
    \endhead
    \multicolumn{2}{r}{\textit{Continue on next page}}\\
    \endfoot
    \endlastfoot
    \begin{swl}#1\end{swl} & \begin{swl}#2\end{swl} \\ \hline
    \textbf{Opportunities} & \textbf{Threats} \\ \hline
    \begin{swl}#3\end{swl} & \begin{swl}#4\end{swl} \\ \hline
  \end{longtable}}

% SWOT table of one app: \appswot{Name}{label-suffix}{S}{W}{O}{T}
\newcommand{\appswot}[6]{%
  \noindent\textbf{Table: SWOT analysis of #1}\label{tab:swot-#2}\par
  \swotbox{#3}{#4}{#5}{#6}}
